# Reaper Education — Lesson 1: Foundations
# Build a number guessing game from scratch

import random

def play_game():
    secret = random.randint(1, 100)
    attempts = 0

    while True:
        guess = int(input("Your guess: "))
        attempts += 1

        if guess == secret:
            print(f"Correct in {attempts} tries!")
            break
        elif guess < secret:
            print("Too low. Keep going.")
        else:
            print("Too high. Try again.")
